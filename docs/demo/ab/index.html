Reproduce

tpt encode --codec cinder --bitrate 160k music-original.wav music-cinder-160k.tpt
tpt decode --container wav music-cinder-160k.tpt music-cinder-160k.wav
tpt compare music-original.wav music-cinder-160k.wav

tpt encode --codec reed --quality 80 speech-original.wav speech-reed.tpt
tpt decode --container wav speech-reed.tpt speech-reed.wav
